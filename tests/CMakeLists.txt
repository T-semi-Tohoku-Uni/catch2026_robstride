cmake_minimum_required(VERSION 3.22)
project(cybergear_host_tests C)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ../Core/Src/main.c)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_callback.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/../Core/Src/main.c" "${CMAKE_CURRENT_BINARY_DIR}/scheduler.inc"
  COMMAND_ERROR_IS_FATAL ANY)
option(CG_SANITIZERS "Enable address/undefined sanitizers with GCC/Clang" OFF)
foreach(rate 100 200)
  add_executable(cg_test_${rate} test_cybergear.c
    ../Core/Src/cybergear.c ../Core/Src/cybergear_position.c
    ../Core/Src/cybergear_controller.c ../Core/Src/cybergear_trajectory.c)
  target_include_directories(cg_test_${rate} PRIVATE stubs ../Core/Inc "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(cg_test_${rate} PRIVATE CYBERGEAR_CONTROL_HZ=${rate})
  set_property(TARGET cg_test_${rate} PROPERTY C_STANDARD 11)
  if(MSVC)
    target_compile_options(cg_test_${rate} PRIVATE /W4 /WX /utf-8)
    if(CG_SANITIZERS)
      string(REPLACE "/RTC1" "" CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG}")
      target_compile_options(cg_test_${rate} PRIVATE /fsanitize=address)
      target_link_options(cg_test_${rate} PRIVATE /INCREMENTAL:NO)
    endif()
  else()
    target_compile_options(cg_test_${rate} PRIVATE -Wall -Wextra -Werror -pedantic)
    target_link_libraries(cg_test_${rate} PRIVATE m)
    if(CG_SANITIZERS)
      target_compile_options(cg_test_${rate} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(cg_test_${rate} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME cg_${rate} COMMAND cg_test_${rate})
endforeach()
